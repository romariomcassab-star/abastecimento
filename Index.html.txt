  async function extractTextFromPdf(file) {
    const arrayBuffer = await file.arrayBuffer();
    const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
    let fullText = "";

    for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
      const page = await pdf.getPage(pageNum);
      const textContent = await page.getTextContent();
      const items = textContent.items.filter(it => it.str && it.str.trim().length > 0);
      
      items.sort((a, b) => {
        const yDiff = b.transform[5] - a.transform[5];
        if (Math.abs(yDiff) > 5) return yDiff;
        return a.transform[4] - b.transform[4];
      });

      let currentLineY = null;
      let pageText = "";

      for (let i = 0; i < items.length; i++) {
        const item = items[i];
        if (currentLineY === null || Math.abs(item.transform[5] - currentLineY) > 5) {
          pageText += (currentLineY === null ? "" : "\n") + item.str;
          currentLineY = item.transform[5];
        } else {
          pageText += " " + item.str;
        }
      }
      fullText += pageText + "\n\n";
    }
    return fullText;
  }

  async function processPdfFile() {
    const fileInput = document.getElementById("pdfFileInput");
    if (!fileInput.files.length) {
      showSysAlert("Atenção", "Por favor, selecione um arquivo PDF do seu computador.", "warning");
      return;
    }

    const file = fileInput.files[0];
    const btn = document.getElementById("btnProcessPdf");
    btn.disabled = true;
    btn.innerHTML = `<span class="spinner-border spinner-border-sm"></span> Processando PDF...`;

    try {
      const extractedText = await extractTextFromPdf(file);
      const response = await apiCall("previewImportacaoTexto", { extractedText: extractedText, fileName: file.name });

      btn.disabled = false;
      btn.innerHTML = `<i class="bi bi-gear-fill"></i> Processar PDF`;

      if (response && response.success) {
        cachedPreviewData = response;

        if (response.duplicidades && response.duplicidades.length > 0) {
          const det = response.duplicidades[0].detalhe || {};
          const dataStr = det.data || 'Data anterior registrada';

          document.getElementById("sysDupDesc").innerHTML = `
            Foram detectados <strong>${response.duplicidades.length} registros</strong> deste planejamento já salvos na planilha.<br><br>
            <div class="text-start bg-white p-2 rounded border small">
              <strong>Arquivo anterior:</strong> ${det.arquivo || 'Desconhecido'}<br>
              <strong>Usuário que importou:</strong> ${det.usuario || 'Usuário anterior'}<br>
              <strong>Data de Importação:</strong> ${dataStr}
            </div>
          `;
          document.getElementById("sysDupMotivoInput").value = "";
          new bootstrap.Modal(document.getElementById("sysDuplicateModal")).show();
        } else {
          isForcingDuplicate = false;
          duplicateForcedReason = "";
          renderPreview(response);

          showSysAlert(
            "PDF Processado com Sucesso!",
            `Foram identificados <strong>${response.totalItens} materiais</strong> distribuídos em <strong>${response.nsasList.length} Pedidos WMS/NSAs</strong>.<br>Conferência liberada na prévia.`,
            "success"
          );
        }

      } else {
        showSysAlert("Não foi possível processar o PDF", response ? response.error : "Falha na leitura.", "danger");
      }

    } catch (err) {
      btn.disabled = false;
      btn.innerHTML = `<i class="bi bi-gear-fill"></i> Processar PDF`;
      showSysAlert("Erro na Leitura", err.message, "danger");
    }
  }

  function updatePreviewItem(index, campo, valor) {
    if (!cachedPreviewData || !cachedPreviewData.itens[index]) return;
    cachedPreviewData.itens[index][campo] = valor.trim();
    const it = cachedPreviewData.itens[index];
    const ok = Boolean(it.codigoProduto && it.lote && it.validade && it.quantidade);
    it.statusLeitura = ok ? "OK" : "PENDENTE_REVISAO";
    const badgeCell = document.getElementById(`statusCell_${index}`);
    if (badgeCell) {
      badgeCell.innerHTML = `<span class="badge ${ok ? 'bg-success' : 'bg-danger'}">${it.statusLeitura}</span>`;
    }
  }

  async function confirmImport() {
    if (!cachedPreviewData) return;

    const btn = document.getElementById("btnConfirmImport");
    btn.disabled = true;
    btn.innerHTML = `<span class="spinner-border spinner-border-sm"></span> Gravando...`;

    try {
      const res = await apiCall("commitImportacao", {
        importData: cachedPreviewData,
        forceDuplicate: isForcingDuplicate,
        duplicateReason: duplicateForcedReason
      });

      btn.disabled = false;
      btn.innerHTML = `Confirmar e Enviar para Impressão`;

      if (res && res.success) {
        showSysAlert("Importação Gravada com Sucesso!", `Foram registrados <strong>${res.total} pedidos</strong> na base de dados com o ID <strong>${res.idImportacao}</strong>.<br>Redirecionando para as Ordens de Abastecimento...`, "success");
        document.getElementById("importPreviewArea").classList.add("d-none");
        document.getElementById("pdfFileInput").value = "";
        document.getElementById("duplicateAlertBox").className = "d-none";
        setTimeout(() => showSection("controle"), 1200);
      } else {
        showSysAlert("Falha ao Gravar", res ? res.error : "Não foi possível persistir os registros.", "danger");
      }
    } catch(e) {
      btn.disabled = false;
      btn.innerHTML = `Confirmar e Enviar para Impressão`;
      showSysAlert("Erro de Gravação", e.message, "danger");
    }
  }

              <small class="d-block text-muted">${g.documento}</small>

      if (res && res.requiresConfirmation) {
        reimpCurrentNsa = nsa;
        document.getElementById("sysReimpDesc").innerText = res.message;
        document.getElementById("sysReimpMotivo").value = "";

    } catch(e) {
      showSysAlert("Erro de Impressão", e.message, "danger");

  }

          <tr>
            <td><small>${h.dataHora}</small></td>
            <td><strong class="text-primary">${h.nsa}</strong></td>
            <td><span class="badge bg-secondary">${h.acao}</span></td>
            <td>${h.statusAnterior} &rarr; <strong>${h.statusNovo}</strong></td>
            <td><small>${h.usuario}</small></td>
            <td><span class="badge bg-dark">${h.area}</span></td>
            <td><small>${h.observacao}</small></td>
          </tr>
        `;
      });

      const inicioFmt = ((res.page - 1) * res.pageSize) + 1;
      const fimFmt = Math.min(res.total, res.page * res.pageSize);
      document.getElementById("auditTotalInfo").innerText = `Exibindo ${inicioFmt}–${fimFmt} de ${res.total} registros (Página ${res.page} de ${res.totalPages})`;

      renderPaginationButtons(res.page, res.totalPages);
    } catch(e) {}
  }

  function renderPaginationButtons(currentPage, totalPages) {
    const ul = document.getElementById("auditPaginationUl");
    ul.innerHTML = "";

    ul.innerHTML += `
      <li class="page-item ${currentPage === 1 ? 'disabled' : ''}">
        <a class="page-link" href="javascript:void(0)" onclick="loadAuditPage(${currentPage - 1})">&laquo;</a>
      </li>
    `;

    let startPage = Math.max(1, currentPage - 2);
    let endPage = Math.min(totalPages, startPage + 4);
    if (endPage - startPage < 4) {
      startPage = Math.max(1, endPage - 4);
    }

    for (let p = startPage; p <= endPage; p++) {
      ul.innerHTML += `
        <li class="page-item ${p === currentPage ? 'active' : ''}">
          <a class="page-link" href="javascript:void(0)" onclick="loadAuditPage(${p})">${p}</a>
        </li>
      `;
    }

    ul.innerHTML += `
      <li class="page-item ${currentPage === totalPages ? 'disabled' : ''}">
        <a class="page-link" href="javascript:void(0)" onclick="loadAuditPage(${currentPage + 1})">&raquo;</a>
      </li>
    `;
  }
</script>
